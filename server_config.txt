# --- 待ち受けるIPアドレス ---
#  空欄のまま = Tailscale の IP を自動で探し、見つからなければ 0.0.0.0。
#  このパソコンだけで使うなら  ip=127.0.0.1  にするのが最も安全です。
#  ⚠️ Tailscale がある環境で空欄にすると、そのアドレスだけで待ち受けます。
#     127.0.0.1 では開けません。起動時に画面へ出るURLを使ってください。
#  ⚠️ 0.0.0.0 は同じネットワークの誰からでも見えます。中身に配慮が要る場合は
#     127.0.0.1 か Tailscale を使ってください（認証機能はありません）。
ip=

# --- 待ち受けポート番号 ---
port=8787
